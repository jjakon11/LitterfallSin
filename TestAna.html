SPdtx <- SPdt %>%
  left_join(Akdt3.1 %>%dplyr::select(sp, Akcon) , by = c("Species"="sp")) %>%
  mutate(Species2=ifelse(is.na(Akcon), "Leafother", Species)) %>%
  group_by(Time, habitat, Species2) %>%
  summarise(Weight=sum(Weight)) %>%
  ungroup() %>%
  group_by(Time, habitat) %>%
  mutate(Wtotal=sum(Weight)) %>%
  ungroup() %>%
  mutate(percent=(Weight/Wtotal)*100) %>%
  left_join(Akdt3.1 %>%dplyr::select(sp, Akcon) , by = c("Species2"="sp")) %>%
  mutate(Concentration=Akcon*percent/100) %>%
  group_by(Time, habitat) %>%
  mutate(Ctotal=sum(Concentration, na.rm=T)) %>%
  ungroup() %>%
  mutate(Akcon2=ifelse(is.na(Akcon), Ctotal, Akcon)) %>% 
  mutate(flux=Weight*Akcon2)

SPBFdt1 <- SPBFdt %>%
  group_by(portion, Species2, habitat) %>%
  summarise(Weight=sum(Weight), 
            flux=sum(flux), 
            Akcon2=mean(Akcon2)) %>%
  ungroup() %>%
  group_by(habitat) %>%
  mutate(Percent=flux/sum(flux)*100) %>%
  ungroup() %>%
  arrange(habitat, desc(Percent))

SPBFdt1_leaf <- SPBFdt1  %>%
  filter(portion=="leaf") %>%
  group_by(habitat) %>%
  mutate(threshod=quantile(Percent, 0.85)) %>%
  mutate(Q=ifelse(Percent>threshod, 1,0)) %>%
  mutate(Percent2= ifelse(Q==1, Percent, sum(Percent[Q==0]) )) %>%
  ungroup() %>%
  mutate(Species2=ifelse(Q==0, "othersp", Species2)) %>%
  dplyr::select(-threshod, -Q) %>%
  group_by(portion, habitat, Species2) %>%
  summarise(Weight=sum(Weight), 
            flux=sum(flux), 
            Akcon2=mean(Akcon2), 
            Percent2=mean(Percent2))%>%
  ungroup() %>%
  arrange(desc(habitat), (Percent2), Species2)

SPBFdt1.0 <- SPBFdt1  %>%

SPBFdt2 <- SPBFdt %>%
  mutate(Time=ifelse(portion!="leaf", "All", Time )) %>%
  group_by(Time, habitat, portion) %>%
  summarise(Weight=sum(Weight)) %>%
  ungroup()


CompoAll2 <- CompoAll %>% 
  filter(Species!="leafother") %>%
  group_by(Time, Cvariable, habitat, portion) %>% 
  summarise(Cvalue=sum(Cvalue)) %>%
  ungroup()

SPBCompo <- CompoAll2 %>%
  left_join(SPBFdt2, by=c("Time"="Time", "habitat"="habitat", "portion"="portion")) %>%
  mutate(flux=Cvalue*Weight) %>%
  group_by(portion, habitat, Cvariable) %>%
  summarise(Cvalue=sum(Cvalue), 
            Weight=sum(Weight),
            flux=sum(flux)) %>%
  ungroup() %>%
  group_by(portion, habitat) %>%
  mutate(Relative=flux/sum(flux)*100) %>%
  ungroup() %>%
  mutate(habitat=ifelse(habitat==0, "windward", 
                        ifelse(habitat==1, "leeward", "creek")))
